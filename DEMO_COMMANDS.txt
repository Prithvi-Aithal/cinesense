-- contype: p = PRIMARY KEY, f = FOREIGN KEY, u = UNIQUE, c = CHECK
SELECT conrelid::regclass AS table_name,
       conname            AS constraint_name,
       CASE contype WHEN 'p' THEN 'PRIMARY KEY' WHEN 'f' THEN 'FOREIGN KEY'
                    WHEN 'u' THEN 'UNIQUE'      WHEN 'c' THEN 'CHECK' END AS type,
       pg_get_constraintdef(oid) AS definition
FROM pg_constraint
WHERE connamespace = 'public'::regnamespace
ORDER BY 1, 3;

-- NOT NULL columns
SELECT table_name, column_name
FROM information_schema.columns
WHERE table_schema = 'public' AND is_nullable = 'NO'
  AND table_name IN ('users','movies','genres','movie_genres','ratings','sessions','rating_audit')
ORDER BY table_name, ordinal_position;

-- Indexes (B-tree for SQL, HNSW for vectors)
SELECT tablename, indexname, indexdef FROM pg_indexes WHERE schemaname = 'public' ORDER BY 1, 2;

-- List all triggers
SELECT tgname AS trigger_name, tgrelid::regclass AS on_table, pg_get_triggerdef(oid) AS definition
FROM pg_trigger WHERE NOT tgisinternal ORDER BY 2, 1;

-- BEFORE / AFTER 1: trg_ratings_audit (AFTER INSERT/UPDATE/DELETE on ratings)
--   Run the whole block: it shows rating_audit before, changes a rating 3 times, shows it after,
--   then rolls back so the demo data stays the same.
BEGIN;
SELECT COUNT(*) AS audit_rows_BEFORE FROM rating_audit;
INSERT INTO ratings (user_id, movie_id, rating)
SELECT u.id, m.id, 4 FROM users u, movies m WHERE u.name = 'Aarav' AND m.title = 'Arrival';
UPDATE ratings SET rating = 2
WHERE user_id = (SELECT id FROM users WHERE name = 'Aarav') AND movie_id = (SELECT id FROM movies WHERE title = 'Arrival');
DELETE FROM ratings
WHERE user_id = (SELECT id FROM users WHERE name = 'Aarav') AND movie_id = (SELECT id FROM movies WHERE title = 'Arrival');
SELECT a.action, u.name AS user_name, m.title, a.old_rating, a.new_rating, a.db_role
FROM rating_audit a LEFT JOIN users u ON u.id = a.user_id LEFT JOIN movies m ON m.id = a.movie_id
ORDER BY a.id DESC LIMIT 3;                                    -- AFTER: INSERT, UPDATE and DELETE rows appeared
ROLLBACK;

-- BEFORE / AFTER 2: trg_ratings_touch (BEFORE UPDATE OF rating) moves rated_at only on a real change
BEGIN;
SELECT m.title, r.rating, r.rated_at AS rated_at_BEFORE
FROM ratings r JOIN movies m ON m.id = r.movie_id
WHERE r.user_id = (SELECT id FROM users WHERE name = 'Aarav') ORDER BY r.movie_id LIMIT 1;
UPDATE ratings SET rating = CASE WHEN rating = 5 THEN 4 ELSE rating + 1 END
WHERE user_id = (SELECT id FROM users WHERE name = 'Aarav')
  AND movie_id = (SELECT MIN(movie_id) FROM ratings WHERE user_id = (SELECT id FROM users WHERE name = 'Aarav'));
SELECT m.title, r.rating, r.rated_at AS rated_at_AFTER
FROM ratings r JOIN movies m ON m.id = r.movie_id
WHERE r.user_id = (SELECT id FROM users WHERE name = 'Aarav') ORDER BY r.movie_id LIMIT 1;
ROLLBACK;

-- Effect 1: trg_ratings_audit wrote these rows (we never INSERTed into rating_audit)
SELECT a.id, a.action, u.name AS user_name, m.title, a.old_rating, a.new_rating, a.db_role,
       to_char(a.changed_at, 'HH24:MI:SS') AS at
FROM rating_audit a
LEFT JOIN users u  ON u.id = a.user_id
LEFT JOIN movies m ON m.id = a.movie_id
ORDER BY a.id DESC
LIMIT 5;

-- Effect 2: re-saving the same rating is NOT logged (the trigger skips it)
BEGIN;
SELECT COUNT(*) AS audit_rows_before FROM rating_audit;
UPDATE ratings SET rating = rating
WHERE (user_id, movie_id) = (SELECT user_id, movie_id FROM ratings ORDER BY user_id, movie_id LIMIT 1);
SELECT COUNT(*) AS audit_rows_after FROM rating_audit;
ROLLBACK;

-- Effect 3: trg_movies_stale_embedding clears the vector when the text changes
BEGIN;
SELECT title, vector_dims(embedding) AS dims FROM movies WHERE title = 'Inception';
UPDATE movies SET description = description || ' (edited)' WHERE title = 'Inception';
SELECT title, embedding IS NULL AS vector_was_cleared FROM movies WHERE title = 'Inception';
ROLLBACK;
SELECT title, vector_dims(embedding) AS dims_after_rollback FROM movies WHERE title = 'Inception';

-- 9a. 3-table JOIN: Sci-Fi movies after 2010
SELECT m.title, m.year
FROM movies m
JOIN movie_genres mg ON mg.movie_id = m.id
JOIN genres g        ON g.id = mg.genre_id
WHERE g.name = 'Sci-Fi' AND m.year > 2010
ORDER BY m.year DESC;

-- 9b. NESTED subquery: users who never rated a Horror movie
SELECT u.name, u.role
FROM users u
WHERE u.id NOT IN (
  SELECT r.user_id
  FROM ratings r
  JOIN movie_genres mg ON mg.movie_id = r.movie_id
  JOIN genres g        ON g.id = mg.genre_id
  WHERE g.name = 'Horror'
)
ORDER BY u.name;

-- 9c. CORRELATED subquery: movies rated 0.5+ above their genre's average
SELECT d.title, g.name AS genre, d.avg_rating,
       (SELECT ROUND(AVG(r2.rating), 2)
        FROM ratings r2 JOIN movie_genres mg2 ON mg2.movie_id = r2.movie_id
        WHERE mg2.genre_id = g.id) AS genre_avg
FROM movie_details d
JOIN movie_genres mg ON mg.movie_id = d.id
JOIN genres g        ON g.id = mg.genre_id
WHERE d.rating_count >= 2
  AND d.avg_rating >= 0.5 + (SELECT AVG(r2.rating)
                             FROM ratings r2 JOIN movie_genres mg2 ON mg2.movie_id = r2.movie_id
                             WHERE mg2.genre_id = g.id)
ORDER BY d.avg_rating DESC, d.title
LIMIT 12;

-- 9d. AGGREGATE + GROUP BY + HAVING + stored function: top rated
SELECT m.title, COUNT(*) AS ratings, ROUND(AVG(r.rating), 2) AS avg,
       bayesian_score(AVG(r.rating), COUNT(*)::int) AS score
FROM ratings r JOIN movies m ON m.id = r.movie_id
GROUP BY m.id
HAVING COUNT(*) >= 2
ORDER BY score DESC
LIMIT 10;

-- 9e. Movies per genre (aggregate)
SELECT g.name AS genre, COUNT(mg.movie_id) AS movies
FROM genres g LEFT JOIN movie_genres mg ON mg.genre_id = g.id
GROUP BY g.name
ORDER BY movies DESC;

-- 9f. RECURSIVE CTE written out in full
WITH RECURSIVE trail (step, id, path, embedding, distance) AS (
  SELECT 0, m.id, ARRAY[m.id], m.embedding, NULL::float8
  FROM movies m WHERE m.title = 'Interstellar'
  UNION ALL
  SELECT t.step + 1, nxt.id, t.path || nxt.id, nxt.embedding, nxt.distance
  FROM trail t
  CROSS JOIN LATERAL (
    SELECT m.id, m.embedding, m.embedding <=> t.embedding AS distance
    FROM movies m
    WHERE m.id <> ALL (t.path) AND m.embedding IS NOT NULL
    ORDER BY m.embedding <=> t.embedding
    LIMIT 1
  ) nxt
  WHERE t.step < 5
)
SELECT t.step, mv.title, ROUND((1 - t.distance)::numeric, 3) AS similarity
FROM trail t JOIN movies mv ON mv.id = t.id
ORDER BY t.step;

-- 9g. HYBRID SQL + VECTOR: recommendations for Aarav (CTE + EXISTS + NOT EXISTS + kNN)
WITH taste AS (
  SELECT AVG(m.embedding) AS v
  FROM ratings r JOIN movies m ON m.id = r.movie_id
  WHERE r.user_id = (SELECT id FROM users WHERE name = 'Aarav') AND r.rating >= 4
)
SELECT m.title,
       ROUND((1 - (m.embedding <=> (SELECT v FROM taste)))::numeric, 3) AS match
FROM movies m
WHERE NOT EXISTS (SELECT 1 FROM ratings r
                  WHERE r.movie_id = m.id AND r.user_id = (SELECT id FROM users WHERE name = 'Aarav'))
  AND EXISTS (SELECT 1 FROM movie_genres mg JOIN genres g ON g.id = mg.genre_id
              WHERE mg.movie_id = m.id AND g.name = 'Thriller')
ORDER BY m.embedding <=> (SELECT v FROM taste)
LIMIT 10;

-- M3. "Write a new trigger" - log every deleted movie into a new table
CREATE TABLE movie_delete_log (
  id         SERIAL PRIMARY KEY,
  movie_id   INT,
  title      TEXT,
  deleted_at TIMESTAMPTZ DEFAULT now()
);
CREATE FUNCTION log_movie_delete() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO movie_delete_log (movie_id, title) VALUES (OLD.id, OLD.title);
  RETURN OLD;
END $$;
CREATE TRIGGER trg_movie_delete_log
AFTER DELETE ON movies
FOR EACH ROW EXECUTE FUNCTION log_movie_delete();

INSERT INTO movies (title, year, description) VALUES ('Trigger Test Movie', 2024, 'Will be deleted.');
DELETE FROM movies WHERE title = 'Trigger Test Movie';
SELECT * FROM movie_delete_log;                                          -- the trigger wrote this row
--   UNDO:
DROP TRIGGER trg_movie_delete_log ON movies;
DROP FUNCTION log_movie_delete();
DROP TABLE movie_delete_log;

-- M4. "Write a BEFORE trigger that blocks something" - ratings cannot be given to movies from the future
CREATE FUNCTION no_future_ratings() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF (SELECT year FROM movies WHERE id = NEW.movie_id) > EXTRACT(YEAR FROM now()) THEN
    RAISE EXCEPTION 'Cannot rate a movie that has not been released yet.';
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER trg_no_future_ratings
BEFORE INSERT ON ratings
FOR EACH ROW EXECUTE FUNCTION no_future_ratings();

BEGIN;
INSERT INTO movies (title, year, description) VALUES ('Future Movie', 2099, 'Not out yet, test only.');
INSERT INTO ratings (user_id, movie_id, rating)
SELECT 1, id, 5 FROM movies WHERE title = 'Future Movie';               -- EXPECTED ERROR from the trigger
ROLLBACK;
--   UNDO:
DROP TRIGGER trg_no_future_ratings ON ratings;
DROP FUNCTION no_future_ratings();

-- M5. "Write a new function" - average rating of a genre
CREATE FUNCTION genre_avg_rating(p_genre text) RETURNS numeric LANGUAGE sql STABLE AS $$
  SELECT ROUND(AVG(r.rating), 2)
  FROM ratings r
  JOIN movie_genres mg ON mg.movie_id = r.movie_id
  JOIN genres g        ON g.id = mg.genre_id
  WHERE g.name = p_genre
$$;
SELECT genre_avg_rating('Sci-Fi') AS scifi, genre_avg_rating('Horror') AS horror;
SELECT name, genre_avg_rating(name) AS avg FROM genres ORDER BY avg DESC NULLS LAST;
--   UNDO:

-- M6. "Write a function that returns a table" - a user's top N movies
CREATE FUNCTION user_top_movies(p_user text, p_n int DEFAULT 3)
RETURNS TABLE (title text, rating smallint) LANGUAGE sql STABLE AS $$
  SELECT m.title::text, r.rating
  FROM ratings r
  JOIN users u  ON u.id = r.user_id
  JOIN movies m ON m.id = r.movie_id
  WHERE u.name = p_user
  ORDER BY r.rating DESC, m.title
  LIMIT p_n
$$;

-- M7. "Write a new procedure" - add a genre to an existing movie
CREATE PROCEDURE add_genre_to_movie(p_title text, p_genre text) LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO genres (name) VALUES (p_genre) ON CONFLICT (name) DO NOTHING;
  INSERT INTO movie_genres (movie_id, genre_id)
  SELECT m.id, g.id FROM movies m, genres g WHERE m.title = p_title AND g.name = p_genre
  ON CONFLICT DO NOTHING;
END $$;
BEGIN;
CALL add_genre_to_movie('Interstellar', 'Mystery');
SELECT title, genres FROM movie_details WHERE title = 'Interstellar';
ROLLBACK;
--   UNDO:
DROP PROCEDURE add_genre_to_movie(text, text);

-- M8. "Write a new query"
-- Directors ranked by average rating (join + group by + having)
SELECT m.director, COUNT(DISTINCT m.id) AS movies, ROUND(AVG(r.rating), 2) AS avg_rating
FROM movies m JOIN ratings r ON r.movie_id = m.id
GROUP BY m.director
HAVING COUNT(r.rating) >= 3
ORDER BY avg_rating DESC
LIMIT 10;

-- Users who rated more movies than the average user (nested subquery in HAVING)
SELECT u.name, COUNT(*) AS rated
FROM users u JOIN ratings r ON r.user_id = u.id
GROUP BY u.name
HAVING COUNT(*) > (SELECT AVG(c) FROM (SELECT COUNT(*) AS c FROM ratings GROUP BY user_id) x)
ORDER BY rated DESC;

-- Movies nobody has rated (NOT EXISTS)
SELECT m.title FROM movies m
WHERE NOT EXISTS (SELECT 1 FROM ratings r WHERE r.movie_id = m.id)
ORDER BY m.title;

-- Highest-rated movie in each genre (window function)
SELECT genre, title, avg_rating FROM (
  SELECT g.name AS genre, d.title, d.avg_rating,
         RANK() OVER (PARTITION BY g.name ORDER BY d.avg_rating DESC NULLS LAST) AS rk
  FROM movie_details d
  JOIN movie_genres mg ON mg.movie_id = d.id
  JOIN genres g        ON g.id = mg.genre_id
) t
WHERE rk = 1
ORDER BY genre;

-- M9. "Change the vector query" - use Euclidean distance (<->) instead of cosine, and top 3
SELECT title, ROUND((embedding <-> (SELECT embedding FROM movies WHERE title = 'Interstellar'))::numeric, 3) AS l2_distance
FROM movies WHERE title <> 'Interstellar'
ORDER BY embedding <-> (SELECT embedding FROM movies WHERE title = 'Interstellar')
LIMIT 3;
