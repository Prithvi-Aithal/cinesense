--   SAY: "\d is PostgreSQL's describe; pg_dump --schema-only regenerates the CREATE statement."
--   Inside psql, \d <table> (4.2) shows the same information.

-- 4.4 Q: "Show the constraints"   contype: p = PRIMARY KEY, f = FOREIGN KEY, u = UNIQUE, c = CHECK
SELECT conrelid::regclass AS table_name,
       conname            AS constraint_name,
       CASE contype WHEN 'p' THEN 'PRIMARY KEY' WHEN 'f' THEN 'FOREIGN KEY'
                    WHEN 'u' THEN 'UNIQUE'      WHEN 'c' THEN 'CHECK' END AS type,
       pg_get_constraintdef(oid) AS definition
FROM pg_constraint
WHERE connamespace = 'public'::regnamespace
ORDER BY 1, 3;

-- Q: "Which columns are NOT NULL?"
SELECT table_name, column_name
FROM information_schema.columns
WHERE table_schema = 'public' AND is_nullable = 'NO'
  AND table_name IN ('users','movies','genres','movie_genres','ratings','sessions','rating_audit')
ORDER BY table_name, ordinal_position;

-- 4.5 Q: "Show your indexes" (B-tree for SQL, HNSW for vectors)   code: db/indexes.sql
SELECT tablename, indexname, indexdef FROM pg_indexes WHERE schemaname = 'public' ORDER BY 1, 2;

-- Q: "What does ON DELETE CASCADE do?" - deleting a movie removes its genre links and ratings
SELECT (SELECT count(*) FROM movie_genres WHERE movie_id = m.id) AS genre_links,
       (SELECT count(*) FROM ratings WHERE movie_id = m.id)      AS ratings
FROM movies m WHERE m.title = 'Demo Film';                                                    -- BEFORE: 2 links, 1 rating
DELETE FROM movies WHERE title = 'Demo Film' RETURNING id, title;
SELECT count(*) AS demo_film_rows FROM movies WHERE title = 'Demo Film';                      -- AFTER: 0
SELECT COUNT(*) AS orphan_genre_links FROM movie_genres mg
WHERE NOT EXISTS (SELECT 1 FROM movies m WHERE m.id = mg.movie_id);                           -- 0: no orphans

DELETE FROM genres WHERE name = 'Documentary' RETURNING id, name;
DELETE FROM users  WHERE name = 'Demo User'   RETURNING id, name;                              -- cascades to sessions + ratings

-- Q: "Does the history survive the delete?" (rating_audit has no foreign keys on purpose)

-- A4 Q: "Aggregate + stored function + NOT EXISTS"  - For You for a user with no 4-star

-- A5 Q: "CTE + nested subqueries + vector"  - Movie page (Interstellar) -> More like this,
--    with "Same genre only" ticked ($2 = true). WITH nn AS = CTE; (SELECT embedding ...) =
--    scalar subqueries; EXISTS joins movie_genres to itself = correlated subquery
--  PS> code -g lib/queries.ts:106      (similar)   run by app/api/movies/[id]/similar/route.ts
WITH nn AS (
  SELECT m.id, m.embedding <=> (SELECT embedding FROM movies WHERE id = :inter) AS distance
  FROM movies m
  WHERE m.id <> :inter AND m.embedding IS NOT NULL
    AND (SELECT embedding FROM movies WHERE id = :inter) IS NOT NULL   -- no vector yet: no neighbours
    AND (NOT true::boolean OR EXISTS (
          SELECT 1 FROM movie_genres a JOIN movie_genres b ON b.genre_id = a.genre_id
          WHERE a.movie_id = m.id AND b.movie_id = :inter))
  ORDER BY m.embedding <=> (SELECT embedding FROM movies WHERE id = :inter)
  LIMIT 6
)
SELECT d.*, ROUND((1 - nn.distance)::numeric, 3) AS similarity
FROM nn
JOIN movie_details d ON d.id = nn.id
ORDER BY nn.distance;

-- A6 Q: "Your most complex query"  - For You page as Aarav (HYBRID SQL + vector)
--    CTE + correlated NOT EXISTS + EXISTS + vector kNN + LEFT JOIN LATERAL ("Because you liked")
--  PS> code -g lib/queries.ts:127      (recommend)   run by app/api/recommendations/route.ts:35
--    $1 is Aarav's taste vector. The website builds it in Node (route.ts:34): every movie he
--    rated, weighted by (rating - 3), added up. This line builds the same vector in SQL:
SELECT SUM(m.embedding * array_fill((r.rating - 3)::real, ARRAY[384])::vector)::text AS taste
FROM ratings r JOIN movies m ON m.id = r.movie_id
WHERE r.user_id = :aarav AND m.embedding IS NOT NULL \gset
WITH nn AS (
  SELECT m.id, m.embedding <=> :'taste'::vector AS distance
  FROM movies m
  WHERE m.embedding IS NOT NULL
    AND NOT EXISTS (SELECT 1 FROM ratings r WHERE r.user_id = :aarav AND r.movie_id = m.id)
    AND (NULL::text IS NULL OR EXISTS (
          SELECT 1 FROM movie_genres mg JOIN genres g ON g.id = mg.genre_id
          WHERE mg.movie_id = m.id AND g.name = NULL))
  ORDER BY m.embedding <=> :'taste'::vector
  LIMIT 12
)
SELECT d.title, d.genres, ROUND((1 - nn.distance)::numeric, 3) AS similarity,
       because.title AS because_you_liked
FROM nn
JOIN movie_details d ON d.id = nn.id
LEFT JOIN LATERAL (
  SELECT lm.id, lm.title
  FROM ratings lr
  JOIN movies lm ON lm.id = lr.movie_id
  WHERE lr.user_id = :aarav AND lr.rating >= 4
  ORDER BY lm.embedding <=> (SELECT embedding FROM movies WHERE id = nn.id)
  LIMIT 1
) because ON true
ORDER BY nn.distance;
--   -> the same movies, in the same order, as the For You page for Aarav

-- B5 Q: "Show the RECURSIVE query written out" (the same as similarity_trail(), A7)
--    (Query Lab: "Similarity trail", lib/lab.ts:99)
WITH RECURSIVE trail (step, id, path, embedding, distance) AS (
  SELECT 0, m.id, ARRAY[m.id], m.embedding, NULL::float8
  FROM movies m WHERE m.title = 'Interstellar'
  UNION ALL
  SELECT t.step + 1, nxt.id, t.path || nxt.id, nxt.embedding, nxt.distance
  FROM trail t
  CROSS JOIN LATERAL (
    SELECT m.id, m.embedding, m.embedding <=> t.embedding AS distance
    FROM movies m
    WHERE m.id <> ALL (t.path) AND m.embedding IS NOT NULL
    ORDER BY m.embedding <=> t.embedding
    LIMIT 1
  ) nxt
  WHERE t.step < 5
)
SELECT t.step, mv.title, ROUND((1 - t.distance)::numeric, 3) AS similarity
FROM trail t JOIN movies mv ON mv.id = t.id
ORDER BY t.step;

-- B6 Q: "Recommendations in ONE SQL statement" - Aarav, Thriller only
--    (Query Lab: "Recommendations, entirely in SQL", lib/lab.ts:140; the website's version is A6)
WITH taste AS (
  SELECT AVG(m.embedding) AS v
  FROM ratings r JOIN movies m ON m.id = r.movie_id
  WHERE r.user_id = (SELECT id FROM users WHERE name = 'Aarav') AND r.rating >= 4
)
SELECT m.title,
       ROUND((1 - (m.embedding <=> (SELECT v FROM taste)))::numeric, 3) AS match
FROM movies m
WHERE NOT EXISTS (SELECT 1 FROM ratings r
                  WHERE r.movie_id = m.id AND r.user_id = (SELECT id FROM users WHERE name = 'Aarav'))
  AND EXISTS (SELECT 1 FROM movie_genres mg JOIN genres g ON g.id = mg.genre_id
              WHERE mg.movie_id = m.id AND g.name = 'Thriller')
ORDER BY m.embedding <=> (SELECT v FROM taste)
LIMIT 10;

  FROM ratings r
  JOIN movie_genres mg ON mg.movie_id = r.movie_id
  JOIN genres g        ON g.id = mg.genre_id
  WHERE g.name = p_genre

-- M8. Q: "Write a new query"
-- Directors ranked by average rating (join + group by + having)
SELECT m.director, COUNT(DISTINCT m.id) AS movies, ROUND(AVG(r.rating), 2) AS avg_rating
FROM movies m JOIN ratings r ON r.movie_id = m.id
GROUP BY m.director
HAVING COUNT(r.rating) >= 3
ORDER BY avg_rating DESC
LIMIT 10;

-- Users who rated more movies than the average user (nested subquery in HAVING)
SELECT u.name, COUNT(*) AS rated
FROM users u JOIN ratings r ON r.user_id = u.id
GROUP BY u.name
HAVING COUNT(*) > (SELECT AVG(c) FROM (SELECT COUNT(*) AS c FROM ratings GROUP BY user_id) x)
ORDER BY rated DESC;

-- Movies nobody has rated (NOT EXISTS)
SELECT m.title FROM movies m
WHERE NOT EXISTS (SELECT 1 FROM ratings r WHERE r.movie_id = m.id)
ORDER BY m.title;

-- Highest-rated movie in each genre (window function)
SELECT genre, title, avg_rating FROM (
  SELECT g.name AS genre, d.title, d.avg_rating,
         RANK() OVER (PARTITION BY g.name ORDER BY d.avg_rating DESC NULLS LAST) AS rk
  FROM movie_details d
  JOIN movie_genres mg ON mg.movie_id = d.id
  JOIN genres g        ON g.id = mg.genre_id
) t
WHERE rk = 1
ORDER BY genre;

SELECT title, ROUND((embedding <-> (SELECT embedding FROM movies WHERE title = 'Interstellar'))::numeric, 3) AS l2_distance
FROM movies WHERE title <> 'Interstellar'
ORDER BY embedding <-> (SELECT embedding FROM movies WHERE title = 'Interstellar')
LIMIT 3;

--    touches the database.  * pgcrypto: bcrypt passwords inside the database.
--  * HNSW index: no training step, fast, high recall on small data.

CREATE OR REPLACE VIEW movie_details AS
SELECT m.id, m.title, m.year, m.director, m.description,
       COALESCE(g.genres, '{}')         AS genres,
       ROUND(AVG(r.rating)::numeric, 2) AS avg_rating,
       COUNT(r.rating)::int             AS rating_count,
       m.runtime_min
FROM movies m
LEFT JOIN LATERAL (
  SELECT array_agg(gn.name ORDER BY gn.name) AS genres
  FROM movie_genres mg JOIN genres gn ON gn.id = mg.genre_id
  WHERE mg.movie_id = m.id
) g ON true
LEFT JOIN ratings r ON r.movie_id = m.id
GROUP BY m.id, g.genres;
